{{- /* Images from the vault are resized to WebP in a few widths with srcset. Obsidian's width syntax in the alt text sets the display size: ![foto|300](img.png) or ![foto|300x200](img.png). */ -}} {{- $alt := .PlainText -}} {{- $width := 0 -}} {{- $title := .Title -}} {{- with findRESubmatch `^(.*?)\|\s*(\d+)(?:x\d+)?\s*$` $alt 1 -}} {{- $alt = index (index . 0) 1 -}} {{- $width = int (index (index . 0) 2) -}} {{- end -}} {{- /* A YouTube video (![](https://youtube.com/watch?v=...), as Obsidian embeds it): the player, from youtube-nocookie.com. With the cookie banner, a button loads it until the visitor accepts videos (consent.js). */ -}} {{- $yt := "" -}} {{- with findRESubmatch `^https?://(?:www\.|m\.)?(?:youtube\.com/(?:watch\?(?:.*&)?v=|shorts/|embed/|live/)|youtu\.be/)([\w-]{11})` .Destination 1 -}} {{- $yt = index (index . 0) 1 -}} {{- end -}} {{- if $yt -}} {{- $src := printf "https://www.youtube-nocookie.com/embed/%s" $yt -}} {{- $ask := false }}{{ with site.Params.blup.cookies }}{{ $ask = .banner }}{{ end -}} {{- if $ask -}} {{ T "ckVideoNote" }} {{- else -}} {{- end -}} {{- else -}} {{- with partial "vault-resource.html" (dict "page" .Page "path" .Destination) -}} {{- if ne .ResourceType "image" -}} {{- /* An embedded PDF, audio or video: the player, or a link. */ -}} {{- if eq .MediaType.MainType "audio" -}} {{- else if eq .MediaType.MainType "video" -}} {{- else -}} {{ $alt | default (path.Base .Name) }} {{- end -}} {{- else if or (in (slice "image/svg+xml" "image/gif") .MediaType.Type) (not (reflect.IsImageResourceProcessable .)) -}} {{- else -}} {{- $img := . -}} {{- $srcset := slice -}} {{- $fallback := "" -}} {{- range $w := slice 480 800 1200 1600 -}} {{- if le $w $img.Width -}} {{- $r := $img.Resize (printf "%dx webp q80" $w) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink $w) -}} {{- if not $fallback }}{{ $fallback = $r }}{{ end -}} {{- end -}} {{- end -}} {{- if not $fallback -}}{{- $fallback = $img.Process "webp q80" -}}{{- $srcset = slice (printf "%s %dw" $fallback.RelPermalink $img.Width) -}}{{- end -}} {{- $h := cond (gt $width 0) (div (mul $img.Height $width) $img.Width) $img.Height -}} {{- end -}} {{- else -}} {{- end -}} {{- end -}}
