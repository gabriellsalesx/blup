{{- /* Links between notes. Obsidian writes paths relative to the note ("../Pasta/Nota%20X.md#titulo"); turn them into the page URL. Attachments (PDF, audio...) are published from the vault mount. External links get rel="noopener" and open normally. */ -}} {{- $u := urls.Parse .Destination -}} {{- $href := .Destination -}} {{- $ext := false -}} {{- $plain := false -}} {{- if $u.IsAbs -}} {{- $ext = true -}} {{- else if $u.Path -}} {{- $target := path.Join .Page.File.Dir $u.Path -}} {{- /* A translation links to the translated note (it may have its own name: the original's translation in this language), else to the original. */ -}} {{- $main := (index hugo.Sites 0) -}} {{- $lang := site.Language.Lang -}} {{- with or (.Page.GetPage $target) (.Page.GetPage (strings.TrimSuffix ".md" $target)) ($main.GetPage $target) ($main.GetPage (strings.TrimSuffix ".md" $target)) -}} {{- $href = partial "bento/link.html" . -}} {{- range .Translations }}{{ if eq .Language.Lang $lang }}{{ $href = partial "bento/link.html" . }}{{ end }}{{ end -}} {{- else -}} {{- with partial "vault-resource.html" (dict "page" .Page "path" $u.Path) -}} {{- $href = .RelPermalink -}} {{- else -}} {{- /* A note that is not on the site (a draft, "publish: false", "private: true", or missing): plain text, so neither a broken link nor the note's file name is published. */ -}} {{- $note := eq (lower (path.Ext $u.Path)) ".md" -}} {{- if $note }}{{ $plain = true }}{{ end -}} {{- if not (and $note (in (site.Params.blup.unpublished | default slice) (strings.TrimPrefix "/" $target))) -}} {{- warnf "%s: link not found: %s" $.Page.File.Path $.Destination -}} {{- end -}} {{- end -}} {{- end -}} {{- with $u.Fragment }}{{ $href = printf "%s#%s" $href (anchorize .) }}{{ end -}} {{- else if $u.Fragment -}} {{- /* A heading of the same note: "#My%20Part" → "#my-part". */ -}} {{- $href = printf "#%s" (anchorize $u.Fragment) -}} {{- end -}} {{- if $plain -}} {{ .Text }} {{- else -}} {{ .Text }} {{- end -}} {{- /**/ -}}
