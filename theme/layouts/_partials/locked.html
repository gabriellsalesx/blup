{{- /* Password of a protected note: the "password" property. "true", "sim", "sí", "si" or "yes" means the site-wide password set in the CLI. Returns "" when the note is not protected. */ -}} {{- $pw := "" -}} {{- with .Params.password -}} {{- $s := lower (string .) -}} {{- if in (slice "true" "sim" "sí" "si" "yes") $s }}{{ $pw = "__global__" }}{{ else }}{{ $pw = string . }}{{ end -}} {{- end -}} {{- return $pw -}}
