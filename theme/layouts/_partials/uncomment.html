{{- /* Removes Obsidian comments (%% text %%, inline or over several lines) from rendered HTML: they are notes to the writer, hidden in Obsidian's reading view too. Code is kept as written: the first alternative consumes 
/ blocks whole, so a %% inside them never starts or ends a
comment. A "%%" left open hides the rest of the note, as in Obsidian. */ -}}
{{- $h := string . -}}
{{- if strings.Contains $h "%%" -}}
  {{- $h = replaceRE `(?s)(<(?:pre|code)\b.*?)|%%(?:||.)*?(?:%%|\z)` "$1" $h -}}
  {{- /* A comment opened inside a tag (an image's alt text) leaves it
  cut: drop the piece so it does not swallow the page after it. */ -}}
  {{- $h = replaceRE `<[^>]*\z` "" $h -}}
  {{- $h = replaceRE `
(?:\s|)*
` "" $h -}}
{{- end -}}
{{- return $h -}}
