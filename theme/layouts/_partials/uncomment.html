{{- /* Removes Obsidian comments (%% text %%, inline or over several lines) from rendered HTML: they are notes to the writer, hidden in Obsidian's reading view too. Code is kept as written: the first alternative consumes 
/ blocks whole, so a %% inside them never starts a comment. */ -}}
{{- $h := string . -}}
{{- if strings.Contains $h "%%" -}}
  {{- $h = replaceRE `(?s)(<(?:pre|code)\b.*?)|%%.*?%%` "$1" $h -}}
  {{- $h = replaceRE `
(?:\s|)*
` "" $h -}}
{{- end -}}
{{- return $h -}}
