{{- /* The changelog's map: the last twelve months as a grid of days, one column a week, like GitHub's contributions. A day with versions takes the color of what they were (bento/change-kind.html: a fix is red, something new green...), stronger with more of them; hovering shows the day and its versions, clicking opens the version. Built with the site, no requests. Param: the changelog's section page. */ -}} {{- $entries := partial "visible.html" .RegularPagesRecursive -}} {{- $order := slice "breaking" "security" "fix" "new" "improvement" "other" -}} {{- $names := dict "new" (T "clKindNew") "improvement" (T "clKindImprovement") "fix" (T "clKindFix") "security" (T "clKindSecurity") "breaking" (T "clKindBreaking") "other" (T "clKindOther") -}} {{- $today := time.AsTime (now.Format "2006-01-02") -}} {{- $start := $today.AddDate 0 0 (sub -364 (int $today.Weekday)) -}} {{- /* The entries of each day of the period. */ -}} {{- $days := dict -}} {{- $total := 0 -}} {{- range $entries -}} {{- if and (not .Date.IsZero) (ge .Date.Unix $start.Unix) -}} {{- $k := .Date.Format "2006-01-02" -}} {{- $days = merge $days (dict $k ((index $days $k) | default slice | append .)) -}} {{- $total = add $total 1 -}} {{- end -}} {{- end -}}
