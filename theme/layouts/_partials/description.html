{{- /* One-line description of a page: the "description" property, else the start of the text (without the leading title and Obsidian comments). The home falls back to site_description and then to its own text. */ -}} {{- if partial "locked.html" . }}{{ return T "lockedDescription" }}{{ end -}} {{- $d := .Description -}} {{- if and (not $d) .IsHome }}{{ $d = (partialCached "site/settings.html" . "site").description }}{{ end -}} {{- if not $d -}} {{- /* The summary first; when it is all comment, the whole text. */ -}} {{- range cond .IsHome (slice (string .Content)) (slice (string .Summary) (string .Content)) -}} {{- if not $d -}} {{- $h := replaceRE `(?s)^\s*]*>.*?` "" . 1 -}} {{- $h = partial "uncomment.html" $h -}} {{- /* A summary can end inside a long comment: a "%%" left open hides the rest, as in Obsidian (code is kept as written). */ -}} {{- $h = replaceRE `(?s)(<(?:pre|code)\b.*?)|%%.*$` "$1" $h -}} {{- $h = replaceRE `<[^>]+>` " " $h -}} {{- $d = $h | htmlUnescape | strings.TrimSpace -}} {{- end -}} {{- end -}} {{- end -}} {{- $d = replaceRE `\s+` " " $d | truncate 160 -}} {{- return $d -}}
