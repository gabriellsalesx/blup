{{- /* The "menu" property of the home note, one line per item, in order: - Docs a folder or note of the site, by name - Guides | Docs "name shown | folder or note" - Shop | https://shop.com a link (http, https, mailto or /path) - Resources | opens the submenu below it, no link - + Guides | Docs "+ " in front: in the submenu of the line above "Home"/"Início" is the home page. Returns a slice of dict name, url, current, external and children (the submenu: the same dicts; the entry then has no url). Empty when the note has no menu. Param: the current page. */ -}} {{- $page := . -}} {{- $lines := slice -}} {{- range site.Home.Params.menu }}{{ $lines = $lines | append (strings.TrimSpace (string .)) }}{{ end -}} {{- $out := slice -}} {{- range $i, $line := $lines -}} {{- if or (eq $i 0) (not (hasPrefix $line "+")) -}} {{- /* The submenu: the "+" lines right below. */ -}} {{- $kids := slice -}} {{- $done := false -}} {{- range $j, $l := $lines -}} {{- if and (gt $j $i) (not $done) -}} {{- if hasPrefix $l "+" -}} {{- with partial "site/menu-link.html" (dict "page" $page "line" (strings.TrimSpace (strings.TrimPrefix "+" $l))) }}{{ $kids = $kids | append . }}{{ end -}} {{- else }}{{ $done = true }}{{ end -}} {{- end -}} {{- end -}} {{- $line = strings.TrimSpace (strings.TrimPrefix "+" $line) -}} {{- if $kids -}} {{- $cur := false -}} {{- range $kids }}{{ if .current }}{{ $cur = true }}{{ end }}{{ end -}} {{- $name := strings.TrimSpace (index (split $line "|") 0) -}} {{- $out = $out | append (dict "name" $name "url" "" "current" $cur "external" false "children" $kids) -}} {{- else -}} {{- with partial "site/menu-link.html" (dict "page" $page "line" $line) }}{{ $out = $out | append . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
