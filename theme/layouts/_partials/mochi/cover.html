"http") -}} {{- with partial "vault-resource.html" (dict "page" $ "path" .) -}} {{- if eq .ResourceType "image" }}{{ $r = . }}{{ end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $r -}}
