{{- /* Top menu entries chosen in the CLI (menus.main), or, when none were chosen, Home plus the top-level folders of the site. Returns a slice of dicts {name, url, current}. */ -}} {{- $page := . -}} {{- $out := slice -}} {{- with partial "site/menu.html" . }}{{ $out = . }}{{ else }} {{- with site.Menus.main -}} {{- range . -}} {{- $url := .URL -}} {{- with .Page }}{{ $url = .RelPermalink }}{{ end -}} {{- $cur := or ($page.IsMenuCurrent "main" .) ($page.HasMenuCurrent "main" .) -}} {{- $out = $out | append (dict "name" .Name "url" $url "current" $cur "external" (hasPrefix $url "http")) -}} {{- end -}} {{- else -}} {{- /* A Bento section the home shows leads home: then it stands for Home itself (bento/link.html). */ -}} {{- $home := site.Home.RelPermalink -}} {{- $secs := slice -}} {{- $homeShown := false -}} {{- range site.Sections.ByWeight -}} {{- $url := partial "bento/link.html" . -}} {{- if eq $url $home }}{{ $homeShown = true }}{{ end -}} {{- $secs = $secs | append (dict "name" (.LinkTitle | default (partial "title.html" .)) "url" $url "current" (or (eq $page .) ($page.IsDescendant .) (and $page.IsHome (eq $url $home)))) -}} {{- end -}} {{- if not $homeShown }}{{ $out = $out | append (dict "name" (T "home") "url" $home "current" $page.IsHome) }}{{ end -}} {{- $out = $out | append $secs -}} {{- end -}} {{- end }} {{- return $out -}}
